/ KEY INPUT EXAMPLE
/ PRESS - OR + TO CHANGE VALUE
/ PRESS ANY KEY TO SEE VALUE
/ ESC to exit

SET C to 128
CALL 4

LABEL 1     / main loop

KEY K         / sets @ to key code or 0

CALLIF 2 43 K
CALLIF 3 45 K
CALL 5
GOTO 1


LABEL 2         / increase value
ADD 1 to C
CALL 4
RET

LABEL 3         / decrease value
SUB 1 from C
CALL 4
RET



LABEL 4         / print out counter value
TABTO 14 12
PRINTNUM C
SPC
SPC
RET


LABEL 5         / print key code and key
TABTO 10 10
VDP K
TABTO 14 10
PRINTNUM K
SPC
SPC
RET



EXIT
